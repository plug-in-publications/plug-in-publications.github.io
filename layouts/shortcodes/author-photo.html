{{/* Round author portrait from a page resource. Usage: {{< author-photo src="featured.jpg" alt="Portrait of ..." >}} */}} {{- $src := .Get "src" | default "featured.jpg" -}} {{- $alt := .Get "alt" | default .Page.Title -}} {{- with .Page.Resources.GetMatch $src -}} {{- $img := .Fill "480x480 Center q85" -}} {{- else -}} {{- errorf "author-photo: resource %q not found in %s" $src .Page.File.Path -}} {{- end -}}
